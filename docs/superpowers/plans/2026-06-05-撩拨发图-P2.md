```typescript
/**
 * M18 撩拨发图 · 分身先撩后发(免费图)编排
 * 客户要图 → 撩拨文字 → 挑免费图 → 发图(type:image, isAiAlter) → 入账。
 * 意图/撩拨走 LLM,无 key 优雅降级(意图默认不发;撩拨用兜底文案)。全程不抛。
 */
import type { Database } from '@loverush/db';
import { pickFreshMedia, imageCooldownOk, recordMediaSend } from './chatMedia';
import { sendMessage } from './chat';

export interface CompanionMediaContext { db: Database }

const TEASE_FALLBACK = '想看啊？那你得先好好哄哄我~';

export interface TeaseFlowArgs {
  conversationId: string;
  customerId: string;
  therapistUserId: string;
  intimacyLevel: number;
  cooldownMessages: number;
  /** 测试/已判定时直接给意图;省略则内部 detectPhotoIntent(customerText) */
  wantPhotoOverride?: boolean;
  customerText?: string;
}

/** 客户最近一句是否在要图 · LLM 判定 · 无 key/失败 → false(不发) */
async function detectPhotoIntent(customerText: string | undefined): Promise<boolean> {
  if (!customerText) return false;
  // 关键词快路 + (可选)LLM。本期:关键词命中即 true,避免额外 LLM 成本与 key 依赖。
  return /照片|图|自拍|看看你|发张|p\b|pic|photo|长什么样|真人/i.test(customerText);
}

async function generateTease(): Promise<string> {
  // 真实撩拨文字应走 reply LLM(人设+T1)。本期兜底常量,Phase 2.1 接 LLM。
  return TEASE_FALLBACK;
}

export async function runTeasePhotoFlow(ctx: CompanionMediaContext, a: TeaseFlowArgs): Promise<void> {
  try {
    const want = a.wantPhotoOverride ?? (await detectPhotoIntent(a.customerText));
    if (!want) return;
    if (!(await imageCooldownOk(ctx, {
      therapistUserId: a.therapistUserId, customerId: a.customerId,
      conversationId: a.conversationId, cooldownMessages: a.cooldownMessages,
    }))) return; // 连发冷却没到 → 这次不发(分身的文字回复已由 ai_alter 发了)
    const media = await pickFreshMedia(ctx, {
      therapistUserId: a.therapistUserId, customerId: a.customerId,
      intimacyLevel: a.intimacyLevel, tiers: ['free'], // 2a 只免费图;付费图 2b
    });
    if (!media) return; // 发光了 → 不重发

    // 1) 撩拨文字(分身身份)
    await sendMessage({ db: ctx.db }, {
      conversationId: a.conversationId, senderUserId: a.therapistUserId,
      content: await generateTease(), type: 'text', isAiAlter: true,
    });
    // 2) 发图
    await sendMessage({ db: ctx.db }, {
      conversationId: a.conversationId, senderUserId: a.therapistUserId,
      content: media.url, type: 'image', isAiAlter: true,
    });
    await recordMediaSend(ctx, {
      therapistUserId: a.therapistUserId, customerId: a.customerId, mediaId: media.id, conversationId: a.conversationId,
    });
  } catch (err) {
    console.warn('[companionMedia] tease photo flow failed (non-fatal):', (err as Error)?.message);
  }
}
```
> 注：`sendMessage` 真实签名以 `apps/api/src/services/chat.ts` 为准(ctx 形态、字段名 isAiAlter/content/type、是否要 customerLocale 等)——读它对齐,去掉不符的字段。`conversations` import 仅测试用。`detectPhotoIntent` 用关键词(避 LLM key 依赖);真实 LLM 意图/撩拨留 Phase 2.1。
