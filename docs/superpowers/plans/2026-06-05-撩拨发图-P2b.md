`apps/api/src/services/chatMediaUnlock.ts`:
```typescript
/**
 * M18 撩拨发图 · 付费私密图解锁(计费交易)
 * 扣客户心动值 → 技师分成 → 发真图 → 入账(防重)。幂等:已解锁过不重复扣。
 */
import { and, eq } from 'drizzle-orm';
import type { Database } from '@loverush/db';
import { chatMedia, chatMediaSends } from '@loverush/db';
import { HttpError } from '../middleware/errors';
import { ErrorCode } from '@loverush/types';
import { debit, credit } from './points';
import { sendMessage } from './chat';
import { recordMediaSend } from './chatMedia';

const MEDIA_SHARE_BPS = 7000; // 技师分成 70%(待 Tony 调,配置化留后续)

export interface UnlockArgs { customerId: string; mediaId: string; conversationId: string; }
export interface UnlockResult { imageUrl: string; pricePoints: number; alreadyUnlocked: boolean; }

export async function unlockChatMedia(ctx: { db: Database }, a: UnlockArgs): Promise<UnlockResult> {
  const [media] = await ctx.db
    .select({ id: chatMedia.id, url: chatMedia.url, pricePoints: chatMedia.pricePoints, therapistUserId: chatMedia.therapistUserId, isActive: chatMedia.isActive })
    .from(chatMedia).where(eq(chatMedia.id, a.mediaId)).limit(1);
  if (!media || media.isActive !== 1) throw HttpError.notFound(ErrorCode.E0003_RESOURCE_NOT_FOUND, 'media not found');

  // 已解锁过 → 幂等:不重复扣,直接发(或视为已发)
  const [sent] = await ctx.db
    .select({ id: chatMediaSends.id })
    .from(chatMediaSends)
    .where(and(eq(chatMediaSends.therapistUserId, media.therapistUserId), eq(chatMediaSends.customerId, a.customerId), eq(chatMediaSends.mediaId, media.id)))
    .limit(1);
  if (sent) return { imageUrl: media.url, pricePoints: media.pricePoints, alreadyUnlocked: true };

  const idem = `companion-media.${a.customerId}.${media.id}`;
  // 扣费(余额不足 debit 内部抛 E2010)
  await debit({ db: ctx.db }, {
    userId: a.customerId, type: 'CHAT_SPEND', amount: media.pricePoints,
    description: `companion-media:${media.id}`, relatedUserId: media.therapistUserId,
    idempotencyKey: idem, metadata: { scene: 'companion_media', mediaId: media.id },
  });
  // 技师分成
  const share = Math.floor((media.pricePoints * MEDIA_SHARE_BPS) / 10000);
  if (share > 0) {
    await credit({ db: ctx.db }, {
      userId: media.therapistUserId, type: 'CHAT_EARN', amount: share,
      description: `companion-media:${media.id}`, relatedUserId: a.customerId,
      idempotencyKey: `${idem}.share`, metadata: { scene: 'companion_media', mediaId: media.id },
    });
  }
  // 发真图(分身身份)
  await sendMessage({ db: ctx.db }, {
    conversationId: a.conversationId, senderUserId: media.therapistUserId,
    text: media.url, type: 'image', isAiAlter: true,
  });
  await recordMediaSend({ db: ctx.db }, { therapistUserId: media.therapistUserId, customerId: a.customerId, mediaId: media.id, conversationId: a.conversationId });
  return { imageUrl: media.url, pricePoints: media.pricePoints, alreadyUnlocked: false };
}
```
> 注：debit/credit/sendMessage 入参字段名以各自 service 实际签名为准(debit 见 companion.ts 用法;sendMessage `text` 非 content)。

`apps/api/src/routes/companionMedia.ts`:
```typescript
import { Hono } from 'hono';
import { zValidator } from '@hono/zod-validator';
import { z } from 'zod';
import { requireAuth } from '../middleware/auth';
import { getDb } from '../db';
import { unlockChatMedia } from '../services/chatMediaUnlock';

export const companionMediaRoutes = new Hono();
companionMediaRoutes.use('*', requireAuth);

const Body = z.object({ conversation_id: z.string().uuid() });

companionMediaRoutes.post('/:mediaId/unlock', zValidator('json', Body), async (c) => {
  const customerId = c.get('userId') as string;
  const mediaId = c.req.param('mediaId');
  const { conversation_id } = c.req.valid('json');
  const res = await unlockChatMedia({ db: getDb() }, { customerId, mediaId, conversationId: conversation_id });
  return c.json({ data: res });
});
```
